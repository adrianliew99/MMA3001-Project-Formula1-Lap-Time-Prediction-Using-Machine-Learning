14def seconds(values): 

15 """Convert timedelta-like values to total seconds; missing values stay NaN.""" 

16 return pd.to_timedelta(values, errors="raise").dt.total_seconds()

19def prepare_laps(raw, session_id): 

20 """Copy source rows, add seconds and flags; never silently remove observations. 

21 

22 Conservative eligibility requires all quality fields to be present and known. 

23 TrackStatus '1' means only green running. Dry compounds are SOFT/MEDIUM/HARD. 

24 """ 

25 d = pd.DataFrame(raw).copy().reset_index(drop=True) 

26 for col in ["Driver", "LapNumber", "LapTime"]: 

27 if col not in d: 

28 raise ValueError(f"Required source field absent: {col}") 

29 d["SourceRow"] = np.arange(len(d)) 

30 d["SessionID"] = session_id 

31 for col in ["LapTime", "Sector1Time", "Sector2Time", "Sector3Time"]: 

32 d[col + "Seconds"] = seconds(d[col]) if col in d else np.nan 

33 for col in QUALITY + ["Compound", "TyreLife", "Stint"]: 

34 if col not in d: 

35 d[col] = pd.NA 

36 d["FlagDuplicate"] = d.duplicated(KEYS, keep=False) 

37 d["FlagInvalidID"] = d[KEYS].isna().any(axis=1) | d.LapNumber.le(0) | d.LapNumber.mod(1).ne(0) 

38 d["FlagInvalidTime"] = ~np.isfinite(d.LapTimeSeconds) | d.LapTimeSeconds.le(0) 

39 d["FlagPit"] = d.PitInTime.notna() | d.PitOutTime.notna() 

40 d["FlagInterrupted"] = d.TrackStatus.astype("string").ne("1").fillna(True) 

41 d["FlagInaccurate"] = ~d.IsAccurate.eq(True).fillna(False) 

42 d["FlagDeleted"] = d.Deleted.eq(True).fillna(False) 

43 d["FlagGenerated"] = d.FastF1Generated.eq(True).fillna(False) 

44 unknown = pd.Series(False, index=d.index) 

45 for col in ["TrackStatus", "IsAccurate", "Deleted", "FastF1Generated"]: 

46 unknown |= d[col].isna() 

47 if any(col not in raw.columns for col in ["PitInTime", "PitOutTime"]): 

48 unknown[:] = True 

49 d["FlagUnknownQuality"] = unknown 

50 d["FlagUnsupportedTyre"] = ~d.Compound.isin(["SOFT", "MEDIUM", "HARD"]) | d.TyreLife.isna() | d.TyreLife.lt(0) | d.Stint.isna() 

51 flags = [c for c in d if c.startswith("Flag")] 

52 d["Eligible"] = ~d[flags].any(axis=1) 

53 return d

56def audit_laps(d, source_columns=None): 

57 """Return JSON-safe counts, overlapping flags, and sequential exclusion effects.""" 

58 keep = pd.Series(True, index=d.index) 

59 effects = [] 

60 for flag in [c for c in d if c.startswith("Flag")]: 

61 before = int(keep.sum()) 

62 keep &= ~d[flag] 

63 effects.append({"rule": flag, "flagged": int(d[flag].sum()), "newly_excluded": before-int(keep.sum()), "remaining": int(keep.sum())}) 

64 return {"rows": len(d), "drivers": int(d.Driver.nunique()), "eligible_rows": int(d.Eligible.sum()), 

65 "source_columns": list(source_columns if source_columns is not None else d.columns), 

66 "unavailable_quality_fields": [c for c in QUALITY if source_columns is not None and c not in source_columns], 

67 "missing": {c: int(d[c].isna().sum()) for c in d.columns}, 

68 "nonpositive_durations": {c: int(d[c].le(0).sum()) for c in d if c.endswith("Seconds")}, 

69 "lap_coverage": d.groupby("Driver").LapNumber.agg(["min", "max", "count", "nunique"]).to_dict("index"), 

70 "exclusion_waterfall": effects}

73def collect_session(root, year=2024, event="Bahrain", session="R"): 

74 """Load only laps and optional weather; export source CSV, audit, and manifest. 

75 

76 Raises on failed/empty lap retrieval. No synthetic fallback is permitted. 

77 FastF1 caching reduces repeat network requests. Weather is not joined to laps. 

78 """ 

79 import fastf1 

80 root = Path(root) 

81 for folder in ["cache", "data/raw", "data/processed", "reports", "results/figures", "results/tables"]: 

82 (root / folder).mkdir(parents=True, exist_ok=True) 

83 fastf1.Cache.enable_cache(str(root / "cache")) 

84 s = fastf1.get_session(year, event, session) 

85 s.load(laps=True, telemetry=False, weather=True, messages=True) 

86 raw = pd.DataFrame(s.laps).copy() 

87 if raw.empty: 

88 raise RuntimeError("FastF1 returned no laps; no dataset was generated.") 

89 sid = f"{year}-{int(s.event.RoundNumber):02d}-{session}" 

90 raw.to_csv(root / f"data/raw/{sid}_laps.csv", index=False) 

91 prepared = prepare_laps(raw, sid) 

92 prepared["Year"] = year 

93 prepared["Event"] = s.event.EventName 

94 prepared["Session"] = session 

95 prepared["EventDate"] = str(s.event.EventDate) 

96 prepared.to_csv(root / f"data/processed/{sid}_laps.csv", index=False) 

97 weather = getattr(s, "weather_data", None) 

98 weather_rows = 0 

99 if weather is not None and len(weather): 

100 weather_rows = len(weather) 

101 weather.to_csv(root / f"data/raw/{sid}_weather.csv", index=False) 

102 audit = audit_laps(prepared, raw.columns) 

103 (root / f"reports/{sid}_quality.json").write_text(json.dumps(audit, indent=2), encoding="utf-8") 

104 manifest = {"source": "FastF1 / F1 timing data", "session_id": sid, "year": year, "event": str(s.event.EventName), 

105 "session": session, "retrieved_utc": datetime.now(timezone.utc).isoformat(), 

106 "versions": {p: version(p) for p in ["fastf1", "pandas", "numpy"]}, 

107 "rows": len(raw), "weather_rows": weather_rows, 

108 "transformations": "Copy all rows; SourceRow zero-based; timedelta total_seconds; explicit conservative quality flags; no weather join.", 

109 "quality_report": f"reports/{sid}_quality.json"} 

110 (root / f"data/raw/{sid}_manifest.json").write_text(json.dumps(manifest, indent=2), encoding="utf-8") 

111 return prepared, audit, manifest
