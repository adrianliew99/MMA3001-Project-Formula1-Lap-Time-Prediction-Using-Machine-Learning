11def make_examples(laps, history=3): 

12 """Use consecutive clean laps within one driver/stint to predict the next lap. 

13 

14 Eligibility of the target is retrospective: results describe uninterrupted dry 

15 running, not the ability to forecast future incidents or pit stops. 

16 """ 

17 if not isinstance(history, int) or history < 1: 

18 raise ValueError("history must be a positive integer") 

19 if laps.duplicated(["SessionID", "Driver", "LapNumber"]).any(): 

20 raise ValueError("Duplicate lap identifiers must be resolved before modeling") 

21 rows = [] 

22 for (_, driver), g in laps.groupby(["SessionID", "Driver"]): 

23 g = g.sort_values("LapNumber") 

24 for end in range(history, len(g)): 

25 block = g.iloc[end-history:end+1] 

26 if not block.Eligible.all() or block.Stint.nunique() != 1 or not np.all(np.diff(block.LapNumber) == 1): 

27 continue 

28 past, target = block.iloc[:-1], block.iloc[-1] 

29 row = {"SessionID": target.SessionID, "Driver": driver, "TargetLap": target.LapNumber, 

30 "EventDate": target.EventDate, "TargetSeconds": target.LapTimeSeconds, 

31 "CurrentTyreLife": past.iloc[-1].TyreLife, "CurrentLap": past.iloc[-1].LapNumber} 

32 row.update({f"Lag{i+1}": float(past.iloc[-i-1].LapTimeSeconds) for i in range(history)}) 

33 rows.append(row) 

34 return pd.DataFrame(rows)

37def split_weekends(examples, train_ids, validation_ids, test_ids): 

38 """Reject overlap, missing sessions, and nonchronological partitions.""" 

39 groups = [list(x) for x in [train_ids, validation_ids, test_ids]] 

40 if any(not x for x in groups): 

41 raise ValueError("Set nonempty training, validation and test race IDs") 

42 # Remove session suffix: race and sprint from one weekend cannot cross sets. 

43 weekends = [{sid.rsplit("-", 1)[0] for sid in ids} for ids in groups] 

44 if any(weekends[i] & weekends[j] for i in range(3) for j in range(i+1, 3)): 

45 raise ValueError("Race weekends overlap between partitions") 

46 parts = [] 

47 for ids in groups: 

48 if not set(ids).issubset(set(examples.SessionID)): 

49 raise ValueError("A configured session has no usable examples") 

50 parts.append(examples[examples.SessionID.isin(ids)].copy()) 

51 dates = [pd.to_datetime(p.EventDate) for p in parts] 

52 if not (dates[0].max() < dates[1].min() and dates[1].max() < dates[2].min()): 

53 raise ValueError("Require training before validation before final testing") 

54 return tuple(parts)

57def score(y, prediction): 

58 """MAE is mean absolute error; RMSE gives larger errors more weight, in seconds.""" 

59 return {"MAE_seconds": float(mean_absolute_error(y, prediction)), 

60 "RMSE_seconds": float(np.sqrt(mean_squared_error(y, prediction)))}

63def train_baseline(train, validation, alphas=(0.1, 1.0, 10.0)): 

64 """Fit scaling on training only; choose ridge penalty on validation MAE only. 

65 

66 Ridge minimizes squared errors plus alpha times squared coefficients. Larger 

67 alpha discourages large coefficients. Test data is deliberately not an input. 

68 """ 

69 if not alphas or any(not np.isfinite(a) or a < 0 for a in alphas): 

70 raise ValueError("Supply finite nonnegative ridge alphas") 

71 features = sorted(c for c in train if c.startswith("Lag")) + ["CurrentTyreLife", "CurrentLap"] 

72 records, fitted = [], [] 

73 for alpha in alphas: 

74 model = make_pipeline(StandardScaler(), Ridge(alpha=alpha, solver="svd")) 

75 start = perf_counter() 

76 model.fit(train[features], train.TargetSeconds) 

77 elapsed = perf_counter()-start 

78 pred = model.predict(validation[features]) 

79 records.append({"alpha": alpha, "fit_seconds": elapsed, **score(validation.TargetSeconds, pred)}) 

80 fitted.append(model) 

81 table = pd.DataFrame(records) 

82 best = int(table.MAE_seconds.argmin()) 

83 return fitted[best], features, table

86def evaluate(model, features, test): 

87 """Compare frozen ridge and previous-lap baseline on exactly the same rows.""" 

88 out = test[["SessionID", "Driver", "TargetLap", "TargetSeconds"]].copy() 

89 start = perf_counter() 

90 out["Ridge"] = model.predict(test[features]) 

91 elapsed = perf_counter()-start 

92 out["PreviousLap"] = test.Lag1.to_numpy() 

93 metrics = pd.DataFrame([{ "model": name, "rows": len(test), **score(test.TargetSeconds, out[name])} for name in ["PreviousLap", "Ridge"]]) 

94 metrics["prediction_seconds"] = [np.nan, elapsed] 

95 return metrics, out
