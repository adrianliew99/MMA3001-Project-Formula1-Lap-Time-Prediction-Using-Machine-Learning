def load_exports(folder: Path) -> pd.DataFrame:
    """Load and concatenate real CSV exports with source provenance."""
    paths = find_exports(folder)
    if not paths:
        raise FileNotFoundError(f'No *next_lap_ml*.csv files found under {folder.resolve()}. Run your existing FastF1 exporter, then set DATA_DIR to its output folder.')
    frames = []
    for path in paths:
        frame = pd.read_csv(path)
        absent = sorted(set(REQUIRED_COLUMNS) - set(frame.columns))
        if absent:
            raise ValueError(f'{path}: required columns missing: {absent}')
        frame['SourceFile'] = str(path.resolve())
        frames.append(frame)
    result = pd.concat(frames, ignore_index=True)
    for col in OPTIONAL_COLUMNS:
        if col not in result.columns:
            result[col] = np.nan if col != 'Session' else 'R'
    result['Session'] = result['Session'].fillna('R').astype(str)
    for col in NUMERIC_FIELDS:
        if col in result:
            result[col] = pd.to_numeric(result[col], errors='coerce')
    for col in ('Event', 'Driver', 'TyreCompound'):
        result[col] = result[col].astype('string').str.strip()
    return result

def audit_pairs(data: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame]:
    """Reject malformed pairs; return (clean rows, exclusion count per reason).

    Only input columns on an explicit allowlist enter training later. The historical
    future lap is used as the regression target and for retrospective selection only.
    """
    df = data.copy()
    duplicated = df.duplicated(KEYS, keep='first')
    reasons = pd.DataFrame(index=df.index)
    reasons['duplicate_key'] = duplicated
    reasons['missing_identifiers'] = df[KEYS].isna().any(axis=1)
    reasons['non_consecutive'] = df['TargetLapNumber'] != df['PreviousLapNumber'] + 1
    reasons['invalid_time'] = ~df['PreviousLapTime_s'].between(30, 300) | ~df['NextLapTime_s'].between(30, 300)
    reasons['invalid_stint'] = ~df['Stint'].between(1, 100)
    reasons['invalid_tyre_age'] = df['TyreAge_laps'].notna() & ~df['TyreAge_laps'].between(0, 100)
    reasons['invalid_speed'] = df['AvgSpeed_kmh'].notna() & ~df['AvgSpeed_kmh'].between(0, 400) | df['MaxSpeed_kmh'].notna() & ~df['MaxSpeed_kmh'].between(0, 400)
    reasons['invalid_percentages'] = df['FullThrottle_pct'].notna() & ~df['FullThrottle_pct'].between(0, 100) | df['Braking_pct'].notna() & ~df['Braking_pct'].between(0, 100)
    reasons['invalid_temperature'] = df['TrackTemp_C'].notna() & ~df['TrackTemp_C'].between(-10, 80)
    if 'DeltaLapTime_s' in df:
        mismatch = df['DeltaLapTime_s'].notna() & ~np.isclose(df['DeltaLapTime_s'].fillna(0), (df['NextLapTime_s'] - df['PreviousLapTime_s']).fillna(0), atol=1e-05)
        reasons['wrong_delta_identity'] = mismatch
    reasons = reasons.fillna(True).astype(bool)
    clean = df.loc[~reasons.any(axis=1)].copy()
    clean['DeltaLapTime_s'] = clean['NextLapTime_s'] - clean['PreviousLapTime_s']
    clean = clean.sort_values(KEYS).reset_index(drop=True)
    report = pd.DataFrame({'reason': list(reasons.columns), 'excluded_rows': reasons.sum().astype(int).values})
    return (clean, report)

def split_by_season(df: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame, pd.DataFrame]:
    """Fixed chronological seasons, with no shared event IDs across splits."""
    train = df.loc[df['Year'].isin(TRAIN_YEARS)].copy()
    val = df.loc[df['Year'].isin(VAL_YEARS)].copy()
    test = df.loc[df['Year'].isin(TEST_YEARS)].copy()
    groups = [set(zip(t['Year'], t['Event'], t['Session'])) for t in (train, val, test)]
    assert not (groups[0] & groups[1] or groups[0] & groups[2] or groups[1] & groups[2]), 'Event leakage'
    if train.empty or val.empty:
        raise ValueError('Insufficient seasons for training/validation. Update TRAIN_YEARS and VAL_YEARS to disjoint, chronological seasons you have actually downloaded; do not row-random-split.')
    if max(TRAIN_YEARS) >= min(VAL_YEARS) or max(VAL_YEARS) >= min(TEST_YEARS):
        raise ValueError('Season order must be TRAIN < VALIDATION < TEST.')
    return (train, val, test)

def make_pipeline(model, numerical: list[str], categorical: list[str]) -> Pipeline:
    """Return preprocessing and regression pipeline fit on training rows only."""
    num = Pipeline([('impute', SimpleImputer(strategy='median', keep_empty_features=True)), ('scale', StandardScaler())])
    cat = Pipeline([('impute', SimpleImputer(strategy='constant', fill_value='UNKNOWN')), ('onehot', OneHotEncoder(handle_unknown='ignore'))])
    prep = ColumnTransformer([('numeric', num, numerical), ('categorical', cat, categorical)])
    return Pipeline([('preprocessor', prep), ('regressor', model)])

def evaluate_split(df: pd.DataFrame, predicted_delta: np.ndarray) -> dict[str, float]:
    """Evaluate reconstructed next-lap seconds and delta seconds."""
    actual_next = df['NextLapTime_s'].to_numpy(dtype=float)
    previous = df['PreviousLapTime_s'].to_numpy(dtype=float)
    predicted_next = previous + np.asarray(predicted_delta, dtype=float)
    return {'MAE_s': mean_absolute_error(actual_next, predicted_next), 'RMSE_s': np.sqrt(mean_squared_error(actual_next, predicted_next)), 'R2_next_lap': r2_score(actual_next, predicted_next), 'MAE_delta_s': mean_absolute_error(df['DeltaLapTime_s'], predicted_delta)}

def prediction_frame(frame: pd.DataFrame, pred_delta: np.ndarray) -> pd.DataFrame:
    """Create auditable per-lap records from a split and delta predictions."""
    keep = ['Year', 'Event', 'Session', 'Driver', 'Stint', 'PreviousLapNumber', 'TargetLapNumber', 'TyreCompound', 'TyreAge_laps', 'PreviousLapTime_s', 'NextLapTime_s']
    out = frame[keep].copy()
    out['PredictedDeltaLapTime_s'] = pred_delta
    out['PredictedNextLapTime_s'] = out['PreviousLapTime_s'] + pred_delta
    out['Residual_s'] = out['NextLapTime_s'] - out['PredictedNextLapTime_s']
    out['AbsoluteError_s'] = out['Residual_s'].abs()
    return out

def good_lap(lap):

    try:
        return pd.notna(lap['LapTime']) and pd.notna(lap['LapNumber']) and (lap['LapTime'].total_seconds() > 0) and (lap['IsAccurate'] == True) and pd.isna(lap['PitInTime']) and pd.isna(lap['PitOutTime']) and (str(lap['TrackStatus']) == '1')
    except (KeyError, TypeError, AttributeError):
        return False

weighted_mean

Mean weighted by time between samples, ignoring invalid samples.

def weighted_mean(values, weights):
    """Mean weighted by time between samples, ignoring invalid samples."""
    valid = values.notna() & np.isfinite(values) & (weights > 0)
    if not valid.any():
        return np.nan
    w = weights[valid]
    return float((values[valid] * w).sum() / w.sum())

telemetry_features

Summarise only the car telemetry belonging to this completed lap.

def telemetry_features(lap):
    """Summarise only the car telemetry belonging to this completed lap."""
    tel = lap.get_car_data()
    if len(tel) < 3:
        return (None, None)
    t = pd.to_timedelta(tel['Time']).dt.total_seconds()
    dt = t.diff().fillna(0).clip(lower=0)
    if dt.sum() <= 0:
        return (None, None)
    speed = pd.to_numeric(tel['Speed'], errors='coerce')
    throttle = pd.to_numeric(tel['Throttle'], errors='coerce')
    brake = pd.to_numeric(tel['Brake'], errors='coerce')
    stats = {'AvgSpeed_kmh': weighted_mean(speed, dt), 'MaxSpeed_kmh': float(speed.max()), 'FullThrottle_pct': 100 * weighted_mean((throttle >= 95).astype(float), dt), 'Braking_pct': 100 * weighted_mean((brake >= 0.5).astype(float), dt)}
    return (stats, tel)

def recent_track_temperature(weather, previous_lap_end):
    """Use only the latest weather reading at or before lap completion."""
    if weather is None or weather.empty or pd.isna(previous_lap_end):
        return np.nan
    past = weather.loc[weather['Time'] <= previous_lap_end]
    if past.empty:
        return np.nan
    val = past.iloc[-1]['TrackTemp']
    return float(val) if pd.notna(val) else np.nan

def extract_session(session, identity):
    """Build legacy pair features, retaining exclusion counts and source sample."""
    raw = session.laps.copy()
    raw['LapTimeSeconds'] = raw['LapTime'].dt.total_seconds()
    round_number = int(session.event.RoundNumber)
    session_id = f"{identity['year']}-{round_number:02d}-R"
    raw['SessionID'] = session_id
    raw['Year'] = identity['year']
    raw['Event'] = identity['event']
    drivers = identity['drivers'] or sorted(raw['Driver'].dropna().unique().tolist())
    rows, sample = ([], None)
    diagnostics = {'candidate_pairs': 0, 'excluded_quality': 0, 'excluded_gap': 0, 'excluded_stint': 0, 'excluded_telemetry': 0, 'accepted_pairs': 0, 'telemetry_errors': [], 'per_driver_pairs': {}}
    for driver in drivers:
        laps = session.laps.loc[session.laps['Driver'] == driver].sort_values('LapNumber')
        count = 0
        for i in range(len(laps) - 1):
            prev, nxt = (laps.iloc[i], laps.iloc[i + 1])
            diagnostics['candidate_pairs'] += 1
            if not (good_lap(prev) and good_lap(nxt)):
                diagnostics['excluded_quality'] += 1
                continue
            if nxt['LapNumber'] != prev['LapNumber'] + 1:
                diagnostics['excluded_gap'] += 1
                continue
            if pd.isna(prev['Stint']) or pd.isna(nxt['Stint']) or nxt['Stint'] != prev['Stint']:
                diagnostics['excluded_stint'] += 1
                continue
            try:
                features, telemetry = telemetry_features(prev)
            except Exception as exc:
                diagnostics['excluded_telemetry'] += 1
                diagnostics['telemetry_errors'].append({'driver': driver, 'lap': int(prev['LapNumber']), 'error': str(exc)})
                continue
            if features is None:
                diagnostics['excluded_telemetry'] += 1
                continue
            previous_time = prev['LapTime'].total_seconds()
            next_time = nxt['LapTime'].total_seconds()
            rows.append({'Year': identity['year'], 'Event': identity['event'], 'Session': 'Race', 'Driver': driver, 'PreviousLapNumber': int(prev['LapNumber']), 'TargetLapNumber': int(nxt['LapNumber']), 'Stint': prev['Stint'], 'PreviousLapTime_s': previous_time, 'TyreCompound': prev['Compound'], 'TyreAge_laps': prev['TyreLife'], 'TrackTemp_C': recent_track_temperature(session.weather_data, prev['Time']), **features, 'NextLapTime_s': next_time, 'DeltaLapTime_s': next_time - previous_time, 'SessionID': session_id, 'RoundNumber': round_number})
            count += 1
            if sample is None:
                sample = telemetry
        diagnostics['per_driver_pairs'][driver] = count
        print(f'{driver}: {count} consecutive clean-lap pairs')
    diagnostics['accepted_pairs'] = len(rows)
    frames = {'all_laps': raw, 'weather': pd.DataFrame(session.weather_data), 'race_results': pd.DataFrame(session.results), 'next_lap_ml': pd.DataFrame(rows, columns=ML_COLUMNS + ['SessionID', 'RoundNumber'])}
    if sample is not None:
        frames['example_raw_telemetry'] = sample
    return (session_id, frames, diagnostics)

def slug(text: str) -> str:
    """Create an ASCII-safe directory token for a race name."""
    import re
    import unicodedata
    text = unicodedata.normalize('NFKD', str(text)).encode('ascii', 'ignore').decode()
    return re.sub('[^a-z0-9]+', '_', text.lower()).strip('_')

def get_schedule(year: int, retries: int=2) -> pd.DataFrame:
    """Fetch the real event schedule, retrying transient provider errors."""
    last_error: Exception | None = None
    for attempt in range(retries + 1):
        try:
            schedule = fastf1.get_event_schedule(year, include_testing=False)
            if schedule is not None and (not schedule.empty) and ('EventName' in schedule):
                return schedule
            last_error = RuntimeError(f'Schedule for {year} is empty or lacks EventName')
        except Exception as exc:
            last_error = exc
        if attempt < retries:
            LOG.warning('Schedule %d attempt %d failed: %s', year, attempt + 1, last_error)
            time.sleep(2 * (attempt + 1))
    raise RuntimeError(f'Cannot resolve schedule for {year}: {last_error}')

def get_races(year: int, filters: list[str]) -> list[dict[str, Any]]:
    """Resolve user-supplied venue/country/name filters to race rounds."""
    schedule = get_schedule(year)
    resolved: list[dict[str, Any]] = []
    matched: set[str] = set()
    for _, event in schedule.iterrows():
        round_no = pd.to_numeric(event.get('RoundNumber'), errors='coerce')
        if pd.isna(round_no) or round_no < 1 or float(round_no) % 1 != 0:
            continue
        event_name = str(event.get('EventName', '')).strip()
        if not event_name or event_name.lower() == 'nan':
            continue
        searchable = ' '.join((str(event.get(k, '')) for k in ('EventName', 'Country', 'Location'))).lower()
        hits = {f for f in filters if f.casefold() in searchable.casefold()}
        if filters and (not hits):
            continue
        matched |= hits
        resolved.append({'year': int(year), 'round': int(round_no), 'event': event_name})
    unknown = set(filters) - matched
    if unknown:
        raise ValueError(f'Unmatched event filters for {year}: {sorted(unknown)}')
    return sorted(resolved, key=lambda item: (item['year'], item['round']))

def validate_race(folder: Path, entry: dict[str, Any], drivers: list[str]) -> tuple[bool, str, Path | None, int]:
    """Verify completion manifest and nonempty ML CSV before reuse/merging.

    Also validate Year, Event, and selected Driver identifiers in the CSV to
    prevent cross-request reuse. The extraction manifest is kept untouched.
    """
    manifests = sorted(folder.glob('*_manifest.json'))
    if len(manifests) != 1:
        return (False, f'expected one completion manifest, found {len(manifests)}', None, 0)
    try:
        manifest = json.loads(manifests[0].read_text(encoding='utf-8'))
    except (OSError, json.JSONDecodeError) as exc:
        return (False, f'cannot read manifest: {exc}', None, 0)
    if str(manifest.get('status', '')).lower() != 'complete':
        return (False, 'manifest status is not complete', None, 0)
    csvs = sorted(folder.glob('*_next_lap_ml.csv'))
    if len(csvs) != 1:
        return (False, f'expected one ML CSV, found {len(csvs)}', None, 0)
    try:
        frame = pd.read_csv(csvs[0], low_memory=False)
    except Exception as exc:
        return (False, f'cannot read ML CSV: {exc}', None, 0)
    required = {'Year', 'Event', 'Driver', 'PreviousLapTime_s', 'NextLapTime_s'}
    absent = required.difference(frame.columns)
    if absent:
        return (False, f'missing ML columns: {sorted(absent)}', None, 0)
    if frame.empty:
        return (False, 'ML CSV contains zero lap pairs', None, 0)
    years = pd.to_numeric(frame['Year'], errors='coerce')
    if not years.eq(entry['year']).all():
        return (False, 'Year column disagrees with requested race', None, 0)
    if not frame['Event'].fillna('').astype(str).str.casefold().eq(entry['event'].casefold()).all():
        return (False, 'Event column disagrees with requested race', None, 0)
    if drivers and (not frame['Driver'].dropna().astype(str).str.upper().isin(drivers).all()):
        return (False, 'Driver column has unrequested drivers', None, 0)
    if (pd.to_numeric(frame['NextLapTime_s'], errors='coerce') <= 0).any():
        return (False, 'nonpositive next-lap time', None, 0)
    return (True, 'verified complete export', csvs[0], len(frame))

        return {**entry, 'Status': 'cached', 'MLRows': rows, 'Reason': reason, 'CSV': str(csv_path), 'Folder': str(folder)}
    if folder.exists() and any(folder.iterdir()) and (not overwrite):
        return {**entry, 'Status': 'incomplete', 'MLRows': 0, 'Reason': f'{reason}; retry with --overwrite or a different --output', 'CSV': '', 'Folder': str(folder)}
    folder.mkdir(parents=True, exist_ok=True)
    cache.mkdir(parents=True, exist_ok=True)
    command = [sys.executable, '-m', EXTRACTOR_MODULE, '--year', str(entry['year']), '--event', entry['event'], '--output', str(folder), '--cache-dir', str(cache)]
    if drivers:
        command.extend(['--drivers', *drivers])
    if offline:
        command.append('--offline')
    if overwrite:
        command.append('--overwrite')
    LOG.info('Launching %s', ' '.join(command))
    try:
        completed = subprocess.run(command, cwd=PROJECT_ROOT, check=False)
    except OSError as exc:
        return {**entry, 'Status': 'failed', 'MLRows': 0, 'Reason': str(exc), 'CSV': '', 'Folder': str(folder)}
    ok, reason, csv_path, rows = validate_race(folder, entry, drivers)
    successful = completed.returncode == 0 and ok
    return {**entry, 'Status': 'success' if successful else 'failed', 'MLRows': rows if successful else 0, 'Reason': reason if successful else f'extractor exit {completed.returncode}; {reason}', 'CSV': str(csv_path) if successful else '', 'Folder': str(folder)}

combine_exports

Combine verified ML CSVs and optional all-laps CSVs, never duplicating races.

def combine_exports(records: list[dict[str, Any]], output: Path) -> int:
    """Combine verified ML CSVs and optional all-laps CSVs, never duplicating races."""
    accepted = [r for r in records if r['Status'] in ('success', 'cached')]
    frames: list[pd.DataFrame] = []
    raw_frames: list[pd.DataFrame] = []
    for record in accepted:
        ml = pd.read_csv(record['CSV'], low_memory=False)
        ml['RoundNumber'] = record['round']
        ml['SessionID'] = f"{record['year']}-{record['round']:02d}-R"
        frames.append(ml)
        raw_files = sorted(Path(record['Folder']).glob('*_all_laps.csv'))
        if len(raw_files) == 1:
            raw = pd.read_csv(raw_files[0], low_memory=False)
            raw['Year'] = record['year']
            raw['Event'] = record['event']
            raw_frames.append(raw)
    if not frames:
        return 0
    combined = pd.concat(frames, ignore_index=True)
    keys = [c for c in ('Year', 'Event', 'Session', 'Driver', 'PreviousLapNumber', 'TargetLapNumber') if c in combined]
    if keys and combined.duplicated(keys).any():
        raise ValueError('Duplicate lap pairs in aggregate; inspect overlapping source exports')
    write_csv(output / 'combined_next_lap_ml.csv', combined)
    if raw_frames:
        write_csv(output / 'combined_all_laps.csv', pd.concat(raw_frames, ignore_index=True))
    return len(combined)

def build_parser() -> argparse.ArgumentParser:
    """Create CLI preserving existing argument names."""

main

Resolve the download budget, execute exports and summarize real outcomes.

def main(argv: list[str] | None=None) -> int:
    """Resolve the download budget, execute exports and summarize real outcomes."""
    args = build_parser().parse_args(argv)
    if args.max_sessions < 1 or args.max_races_per_year < 1:
        raise SystemExit('--max-sessions and --max-races-per-year must be positive')
    drivers = sorted({d.upper() for d in args.drivers})
    filters = list(dict.fromkeys(args.events))
    logging.basicConfig(level=logging.INFO, format='%(levelname)s %(message)s')
    selections: list[dict[str, Any]] = []
    try:
        for year in sorted(set(args.years)):
            selections.extend(get_races(year, filters)[:args.max_races_per_year])
    except (RuntimeError, ValueError) as exc:
        print(f'Schedule resolution failed: {exc}', file=sys.stderr)
        return 2
    if not selections:
        print('No matching races were resolved', file=sys.stderr)
        return 2
    if len(selections) > args.max_sessions:
        print(f'Resolved {len(selections)} sessions but limit is {args.max_sessions}. Raise --max-sessions explicitly or reduce --events/--years.', file=sys.stderr)
        return 2
    print(f"Resolved {len(selections)} sessions (limit {args.max_sessions}); drivers={drivers or 'ALL'}", flush=True)
    for item in selections:
        print(f"  {item['year']} round {item['round']}: {item['event']}", flush=True)
    if args.dry_run:
        print('Dry run only: no event data was downloaded or written.')
        return 0
    output = args.output.expanduser().resolve()
    cache = args.cache_dir.expanduser().resolve()
    output.mkdir(parents=True, exist_ok=True)
    run_path = output / 'batch_manifest.json'
    records: list[dict[str, Any]] = []
    write_json(run_path, {'status': 'running', 'started_utc': utc_now(), 'selection': selections, 'drivers': drivers, 'offline': args.offline})
    for entry in selections:
        result = export_one(entry, drivers, output, cache, args.offline, args.overwrite)
        records.append(result)
        print(f"{entry['year']} {entry['event']}: {result['Status']}; {result['Reason']}", flush=True)
        write_csv(output / 'export_summary.csv', pd.DataFrame(records))
    try:
        rows = combine_exports(records, output)
        merge_error = None
    except Exception as exc:
        rows, merge_error = (0, str(exc))
        LOG.error('Aggregate failed: %s', exc)
    good = sum((record['Status'] in ('success', 'cached') for record in records))
    completed = good == len(selections) and merge_error is None and (rows > 0)
    write_json(run_path, {'status': 'complete' if completed else 'incomplete', 'finished_utc': utc_now(), 'selection': selections, 'drivers': drivers, 'completed_sessions': good, 'requested_sessions': len(selections), 'combined_ml_rows': rows, 'merge_error': merge_error, 'records': records})
    print(f"Batch {('complete' if completed else 'incomplete')}: {good}/{len(selections)} usable sessions; combined ML rows={rows:,}; {output}")
    return 0 if completed else 1
